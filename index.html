async function savePlayerProfile(user) { const profileRef = ref( db, "users/" + user.uid + "/profile" ); const profile = { uid: user.uid, email: user.email || "", updatedAt: Date.now() }; try { await set(profileRef, profile); databaseReady = true; dbStatus.textContent = "Player profile saved successfully."; } catch (error) { databaseReady = false; console.error( "Database profile save failed:", error.code, error.message ); dbStatus.textContent = "Profile error: " + error.code; } }
